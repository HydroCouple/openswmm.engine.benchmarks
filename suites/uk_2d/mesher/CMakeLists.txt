cmake_minimum_required(VERSION 3.21)
project(uk_gui_mesher LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT EXISTS "${GUI_SOURCE}/src/mesh/meshgenerator.cpp")
  message(FATAL_ERROR "Set GUI_SOURCE to the openswmm.gui checkout")
endif()
find_package(Qt6 REQUIRED COMPONENTS Core Gui Concurrent)
add_library(predicates STATIC "${GUI_SOURCE}/vendor/predicates/predicates.c")
target_compile_options(predicates PRIVATE -Wno-deprecated-non-prototype)
target_include_directories(predicates PUBLIC "${GUI_SOURCE}/vendor/predicates")
set(mesh_sources meshgenerator meshcdt meshpatch meshquadregion pslgprep breaklinestrips)
add_executable(uk_gui_mesher main.cpp "${GUI_SOURCE}/src/core/editgeometry.cpp")
foreach(name IN LISTS mesh_sources)
  target_sources(uk_gui_mesher PRIVATE "${GUI_SOURCE}/src/mesh/${name}.cpp")
endforeach()
target_include_directories(uk_gui_mesher PRIVATE "${GUI_SOURCE}/include")
target_link_libraries(uk_gui_mesher PRIVATE Qt6::Core Qt6::Gui Qt6::Concurrent predicates)
